<script>
let inventory = JSON.parse(localStorage.getItem('mp_inv') || '[]');
let photoCount = 0;
let voiceEnabled = true;
let videoStream = null;
function toggleVoice() { voiceEnabled = !voiceEnabled; document.getElementById('v-status').innerText = voiceEnabled ? "ON" : "OFF"; }
function speak(text) { if(voiceEnabled && 'speechSynthesis' in window) { window.speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(text); u.lang='it-IT'; window.speechSynthesis.speak(u); } }
async function startCamera() {
try {
if(videoStream) videoStream.getTracks().forEach(t => t.stop());
videoStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } });
document.getElementById('viewport').style.display = 'block';
document.getElementById('video').srcObject = videoStream;
} catch(e) { alert("Camera non accessibile."); }
}
function captureAndDecompose() {
if(videoStream) {
videoStream.getTracks().forEach(t => t.stop());
document.getElementById('viewport').style.display = 'none';
}
photoCount++;
document.getElementById('photo-counter').innerText = 'FOTO: ' + photoCount;
const sector = document.getElementById('sector').value;
const map = { ANT: "Faro Anteriore", POST: "Fanale Posteriore", LAT: "Specchio Retrovisore", MOT: "Alternatore", INT: "Quadro Strumenti" };
document.getElementById('part-type').value = map[sector] || "Ricambio Generic";
checkDuplicate("GEN-HASH-" + photoCount);
speak("Foto acquisita e settore scomposto.");
}
function checkDuplicate(hash) {
const oem = document.getElementById('oem-code').value.trim().toUpperCase();
const loc = document.getElementById('location').value.trim().toLowerCase();
const dupe = inventory.find(i => (oem && i.oem === oem) || (loc && i.location.toLowerCase() === loc));
const alertBox = document.getElementById('duplicate-alert');
if(dupe) {
alertBox.innerText = "⚠️ Attenzione: Possibile doppione rilevato per questo ricambio!";
alertBox.style.display = 'block';
speak("Attenzione, doppione rilevato.");
} else {
alertBox.style.display = 'none';
}
}
function runTriangulation() {
const part = document.getElementById('part-type').value;
const model = document.getElementById('veh-model').value;
document.getElementById('triang-results').innerHTML = 🔎 Triangolazione Web per ${part} ${model}... <a href="https://www.google.com/search?q=${encodeURIComponent(part + ' ' + model + ' oem')}" target="_blank">Apri Ricerca</a>;
speak("Triangolazione avviata.");
}
function addItem() {
const partType = document.getElementById('part-type').value.trim();
if(!partType) { alert("Inserisci il Tipo Ricambio."); return; }
const oem = document.getElementById('oem-code').value.trim().toUpperCase();
const model = document.getElementById('veh-model').value.trim() || "Gen";
const year = document.getElementById('veh-year').value.trim() || "N.D.";
const location = document.getElementById('location').value.trim() || "Non Assegnata";
const isNeutral = !oem;
const barcode = oem || "NTR-" + String(inventory.length + 1).padStart(3, '0');
const certainty = isNeutral ? 70 : 100;
const item = { id: inventory.length + 1, partType, oem, model, year, location, barcode, isNeutral, certainty, print: true };
inventory.push(item);
localStorage.setItem('mp_inv', JSON.stringify(inventory));
render();
speak("Ricambio aggiunto all'inventario.");
}
function render() {
const tbody = document.getElementById('inventory-body');
tbody.innerHTML = inventory.map((i, idx) => <tr class="${i.isNeutral ? 'row-neutral' : ''}"> <td>#${i.id}</td> <td><strong>${i.partType}</strong><br><span style="font-size:0.7rem; color:var(--text-muted);">${i.model} (${i.year})</span></td> <td><span style="font-family:monospace;">${i.barcode}</span><br>${i.isNeutral ? '<span class="badge badge-neutral">NEUTRA</span>' : '<span class="badge badge-high">CODIFICATO</span>'}</td> <td>${i.location}</td> <td><span class="badge ${i.certainty===100?'badge-high':'badge-neutral'}">${i.certainty}%</span></td> <td style="text-align:center;"><input type="checkbox" ${i.print?'checked':''} onchange="togglePrint(${idx}, this.checked)"></td> <td><button class="btn btn-magenta" style="padding:2px 8px; margin:0;" onclick="delItem(${idx})">✕</button></td> </tr>).join('');
}
function togglePrint(idx, val) { inventory[idx].print = val; }
function delItem(idx) { inventory.splice(idx, 1); render(); }
function resetInventory() { inventory = []; localStorage.removeItem('mp_inv'); render(); }
function printSelected() {
const toPrint = inventory.filter(i => i.print);
if(toPrint.length === 0) { alert("Nessun articolo selezionato."); return; }
const printArea = document.getElementById('print-area');
printArea.innerHTML = toPrint.map(i => <div class="label"> <div style="font-size:0.8rem; font-weight:bold; display:flex; justify-content:space-between;"><span>${i.isNeutral ? 'ETICHETTA NEUTRA' : 'MP INVENTORY'}</span><span>LOC: ${i.location}</span></div> <div style="font-size:1.2rem; font-weight:bold; margin:5px 0;">${i.partType}</div> <div style="font-size:0.8rem;">Modello: ${i.model} (${i.year}) | Certezza: ${i.certainty}%</div> <div style="text-align:center; margin-top:5px;"><svg id="bc-${i.id}"></svg></div> </div>).join('');
toPrint.forEach(i => {
JsBarcode(#bc-${i.id}, i.barcode, { format: "CODE128", width: 2, height: 40, displayValue: true });
});
window.print();
}
function openReport() {
const total = inventory.length;
const neutral = inventory.filter(i => i.isNeutral).length;
const missingLoc = inventory.filter(i => i.location === "Non Assegnata").length;
document.getElementById('report-body').innerHTML = <p><strong>Totale Ricambi Caricati:</strong> ${total}</p> <p style="color:var(--neon-amber);"><strong>Etichette Neutre Emesse:</strong> ${neutral}</p> <p style="color:${missingLoc>0?'red':'inherit'};"><strong>Ricambi senza Ubicazione:</strong> ${missingLoc}</p> <p style="margin-top:10px; font-size:0.8rem; color:var(--text-muted);">Il report conferma lo stato dell'inventario reattivo.</p>;
document.getElementById('report-modal').style.display = 'flex';
}
function closeReport() { document.getElementById('report-modal').style.display = 'none'; }
render();
